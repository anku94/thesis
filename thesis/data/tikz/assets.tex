% Place a pic by matching one of its anchors to a target coordinate.
% Requires: \usetikzlibrary{calc}
%
% Usage: \PlacePicByAnchor{pic-spec}{name}{target-coord}{anchor}
%        \PlacePicByAnchor[transform]{pic-spec}{name}{target-coord}{anchor}
%
% Examples:
%   \PlacePicByAnchor{storage}{S}{A-east}{west}     % S-west lands at A-east
%   \PlacePicByAnchor[rotate=45]{rank={1}}{R}{0,0}{center}
%
\newcommand{\PlacePicByAnchor}[5][]{%
  \pgfinterruptboundingbox
  \begin{scope}[opacity=0,#1]
    \pic[name=#3-meas] at (0,0) {#2};
  \end{scope}
  \endpgfinterruptboundingbox
  \path let \p1=(#4), \p2=(#3-meas-#5) in
    coordinate (#3-shift) at (\x1-\x2,\y1-\y2);
  \begin{scope}[shift={(#3-shift)},#1]
    \pic[name=#3] at (0,0) {#2};
  \end{scope}
}
