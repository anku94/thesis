% ASSET: application (BSP app = ranks + fabric + storage)
% Requires pics/rank, pics/fabric, pics/storage
\tikzset{
  stub/.style={line width=1.0pt, fablinkcolor!70},
  applabel/.style={anchor=east, font=\sffamily\normalsize, align=right, text width=3.0cm},
  pics/application/.style={
    code={
      \def\rankgap{0.4cm}
      \def\storagegap{0.2cm}

      \begin{scope}[local bounding box=-bbox]
        % -- Ranks in a row --
        \pic (-r0) at (0, 0) {rank={0}};

        \coordinate (-r1pos) at ([xshift=\rankgap]-r0-east);
        \PlacePicByAnchor{rank={1}}{-r1}{-r1pos}{west}

        \coordinate (-r2pos) at ([xshift=\rankgap]-r1-east);
        \PlacePicByAnchor{rank={2}}{-r2}{-r2pos}{west}

        \coordinate (-r3pos) at ([xshift=\rankgap]-r2-east);
        \PlacePicByAnchor{rank={$\cdots$}}{-r3}{-r3pos}{west}

        % -- Fabric band underneath ranks --
        \coordinate (-fabcenter) at ($(-r0-south)!0.5!(-r3-south) + (0,-0.7)$);
        \pic (-fab) at (-fabcenter) {fabric=2.4};

        % -- Storage cylinders underneath fabric --
        \coordinate (-s0pos) at ($(-fab-south) + (-1.5,-0.9)$);
        \pic (-s0) at (-s0pos) {storage};

        \coordinate (-s1pos) at ([xshift=\storagegap]-s0-east);
        \PlacePicByAnchor{storage}{-s1}{-s1pos}{west}

        \coordinate (-s2pos) at ([xshift=\storagegap]-s1-east);
        \PlacePicByAnchor{storage}{-s2}{-s2pos}{west}

        \coordinate (-s3pos) at ([xshift=\storagegap]-s2-east);
        \PlacePicByAnchor{storage}{-s3}{-s3pos}{west}

        % -- Stubs connecting ranks to fabric --
        \begin{scope}[on background layer]
          \foreach \n in {r0,r1,r2,r3} {
            \draw[stub] (-\n-south) -- (-\n-south|-{-fab-north});
          }

          % -- Stubs connecting fabric to storage --
          \foreach \n in {s0,s1,s2,s3} {
            \draw[stub] (-\n-north) -- (-\n-north|-{-fab-south});
          }
        \end{scope}

        % -- Labels on the left --
        \def\labelgap{0.3cm}
        \path let \p1=(-r0-west), \p2=(-fab-west), \p3=(-s0-west) in
          coordinate (-labelx) at ({min(\x1,\x2,\x3) - \labelgap}, 0);
        \node[applabel] at (-labelx|-{-r0-center}) {\textbf{Application Ranks}\\\emph{O(10--100K)}};
        \node[applabel] at (-labelx|-{-fab-center}) {\textbf{RDMA Fabric}\\\emph{400\,Gbps}};
        \node[applabel] at (-labelx|-{-s0-center}) {\textbf{Parallel Storage} \\ \emph{(Lustre, DAOS)}};
      \end{scope}

      \PicBboxAnchors{-bbox}
    }
  }
}
