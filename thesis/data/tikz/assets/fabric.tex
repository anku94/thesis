% ASSET: fabric (network interconnect)
% Usage: \pic {fabric=<half-width>};  e.g., fabric=1.8 gives width 3.6
\tikzset{
  pics/fabric/.style args={#1}{
    code={
      \def\hw{#1}
      \colorlet{fabclr}{nord2}
      \begin{scope}[local bounding box=-bbox]
        % Band
        \draw[line width=1.0pt, draw=fabclr!70, fill=fabclr!5, rounded corners=1pt]
          (-\hw,-0.18) rectangle (\hw, 0.18);
        % Horizontal traces
        \draw[thin, fabclr!50] (-\hw, 0.06) -- (\hw, 0.06);
        \draw[thin, fabclr!50] (-\hw,-0.06) -- (\hw,-0.06);
        % Vertical ticks (switch crosspoints)
        \pgfmathsetmacro{\tickstep}{0.4}
        \pgfmathsetmacro{\numticks}{floor(\hw/\tickstep)}
        \draw[thin, fabclr!50] (0,-0.18) -- (0,0.18);
        \foreach \i in {1,...,\numticks} {
          \pgfmathsetmacro{\xpos}{\i*\tickstep}
          \draw[thin, fabclr!50] (\xpos,-0.18) -- (\xpos,0.18);
          \draw[thin, fabclr!50] (-\xpos,-0.18) -- (-\xpos,0.18);
        }
      \end{scope}
      \PicBboxAnchors{-bbox}
    }
  },
  pics/fabric/.default={1.8}
}
