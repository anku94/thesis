% ASSET: orcatree (TBON-style tree: 1 root, 2 children, 4 leaves)
% Requires ctlcolor, aggcolor, mpicolor from assets.tex
\tikzset{
  orcatreenode/.style={
    circle, draw, thick, inner sep=1.2pt, minimum size=0.4cm
  },
  pics/orcatree/.style={
    code={
      \begin{scope}[
        grow=up,
        level distance=0.55cm,
        sibling distance=2.0cm,
        every node/.style={orcatreenode, fill=ctlcolor},
        level 1/.style={sibling distance=0.9cm, nodes={fill=aggcolor}},
        level 2/.style={sibling distance=0.45cm, nodes={fill=mpicolor}},
        edge from parent/.style={draw, thick}
      ]
        \node[orcatreenode] (root) {}
          child { node[orcatreenode] (c1) {}
            child { node[orcatreenode] (l1) {} }
            child { node[orcatreenode] (l2) {} }
          }
          child { node[orcatreenode] (c2) {}
            child { node[orcatreenode] (l3) {} }
            child { node[orcatreenode] (l4) {} }
          };
      \end{scope}

      % Anchors from bounding box
      \path (current bounding box.north)  coordinate (-north);
      \path (current bounding box.south)  coordinate (-south);
      \path (current bounding box.east)   coordinate (-east);
      \path (current bounding box.west)   coordinate (-west);
      \path (current bounding box.center) coordinate (-center);
    }
  }
}
