% ASSET: pivottable (partition boundary table)
\tikzset{
  pics/pivottable/.style={
    code={
      \def\cw{1.0}  % cell width
      \def\ch{0.35} % cell height
      % Header row (Rank labels)
      \foreach \i/\lbl in {0/Rank 0, 1/Rank 1, 2/Rank 2, 3/Rank 3} {
        \draw[thick, fill=gray!15]
          ({\i*\cw - 2*\cw}, \ch) rectangle ({(\i+1)*\cw - 2*\cw}, {2*\ch});
        \node[anchor=center, font=\sffamily\scriptsize] at ({(\i+0.5)*\cw - 2*\cw}, {1.5*\ch}) {\lbl};
      }
      % Value row (ranges)
      \foreach \i/\val in {0/{0--0.3}, 1/{0.3--0.4}, 2/{0.4--0.6}, 3/{0.6--1.0}} {
        \draw[thick, fill=white]
          ({\i*\cw - 2*\cw}, 0) rectangle ({(\i+1)*\cw - 2*\cw}, \ch);
        \node[anchor=center, font=\sffamily\scriptsize] at ({(\i+0.5)*\cw - 2*\cw}, {0.5*\ch}) {\val};
      }
      % Anchors
      \coordinate (-north)  at (0, {2*\ch});
      \coordinate (-south)  at (0, 0);
      \coordinate (-east)   at ({2*\cw}, {\ch});
      \coordinate (-west)   at ({-2*\cw}, {\ch});
      \coordinate (-center) at (0, {\ch});
    }
  }
}
