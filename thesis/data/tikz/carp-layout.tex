\begin{document}

% === Styles ===
\tikzset{
  % Query arrow
  qarrow/.style={-{Stealth}, thick, shorten <=0.3cm, shorten >=0.3cm},
  % Query annotation (red highlights and merge lines)
  qannot/.style={nord11, line width=1.8pt},
  % SSTable cell
  sstable/.style={
    draw,
    thick,
    fill=pqcolor!20,
    minimum width=2cm,
    minimum height=0.7cm,
    font=\sffamily\small,
  },
  % U-shaped bucket: bucket={v1/v2/v3}
  pics/bucket/.style args={#1/#2/#3}{code={
    % Parameters
    \def\bucketW{1.5cm}
    \def\bucketH{0.7cm}
    \def\bucketPad{0.2cm}
    \def\bucketLip{0.1cm}

    % Stack of 3 SSTables
    \node[sstable, minimum width=\bucketW, minimum height=\bucketH] (-c1) at (0, 0) {#1};
    \node[sstable, minimum width=\bucketW, minimum height=\bucketH, below=0 of -c1] (-c2) {#2};
    \node[sstable, minimum width=\bucketW, minimum height=\bucketH, below=0 of -c2] (-c3) {#3};

    % Beaker shape behind (on background)
    \begin{scope}[on background layer]
      \draw[thick, fill=nord4!50, rounded corners=3pt]
        % Left lip
        ([xshift=-\bucketPad-\bucketLip, yshift=\bucketLip]-c1.north west)
        -- ([xshift=-\bucketPad, yshift=\bucketLip]-c1.north west)
        % Left side down
        -- ([xshift=-\bucketPad, yshift=-\bucketPad]-c3.south west)
        % Flat bottom
        -- ([xshift=\bucketPad, yshift=-\bucketPad]-c3.south east)
        % Right side up
        -- ([xshift=\bucketPad, yshift=\bucketLip]-c1.north east)
        % Right lip
        -- ([xshift=\bucketPad+\bucketLip, yshift=\bucketLip]-c1.north east);
    \end{scope}

    % Anchors
    \coordinate (-north) at ([yshift=\bucketLip]-c1.north);
    \coordinate (-south) at ([yshift=-\bucketPad]-c3.south);
    \coordinate (-center) at ($(-north)!0.5!(-south)$);

    % Debug: show anchors
    \ifdebugassets
      \foreach \a/\pos in {north/above,south/below,center/right} {
        \fill[red] (-\a) circle[radius=1.5pt];
        \node[font=\tiny,red,\pos,inner sep=1pt] at (-\a) {\a};
      }
    \fi
  }},
  % Stacked SSTables icon (for "retrieve overlapping")
  pics/sststack/.style={code={
    \def\stackW{0.8cm}
    \def\stackH{0.5cm}
    \def\stackOff{0.15cm}

    \begin{scope}[local bounding box=-bbox]
      \node[draw, thick, fill=pqcolor!20, minimum width=\stackW, minimum height=\stackH] (-s1) at (0,0) {};
      \node[draw, thick, fill=pqcolor!20, minimum width=\stackW, minimum height=\stackH]
        (-s2) at (\stackOff, \stackOff) {};
      \node[draw, thick, fill=pqcolor!20, minimum width=\stackW, minimum height=\stackH]
        (-s3) at (2*\stackOff, 2*\stackOff) {};
    \end{scope}

    % Anchors from bounding box
    \coordinate (-north) at (-bbox.north);
    \coordinate (-south) at (-bbox.south);
    \coordinate (-center) at (-bbox.center);
    \coordinate (-west) at (-bbox.west);
    \coordinate (-east) at (-bbox.east);

    % Debug: show anchors
    \ifdebugassets
      \foreach \a/\pos in {north/above,south/below,center/right,west/left,east/right} {
        \fill[red] (-\a) circle[radius=1.5pt];
        \node[font=\tiny,red,\pos,inner sep=1pt] at (-\a) {\a};
      }
    \fi
  }},
}

\begin{tikzpicture}
  \def\bucketSep{2.5cm}

  \pic (b1) at (0,0) {bucket={0-28/0-23/0-25}};
  \node[below=0.2cm of b1-south, font=\sffamily\small] {Partition 0};

  % Time axis (left side)
  \draw[->, thick] ([xshift=-1.5cm]b1-south) -- ([xshift=-1.5cm]b1-north)
    node[midway, rotate=90, anchor=south, font=\sffamily\small] {Time};

  % SSTable annotation
  \node[anchor=north, font=\sffamily\footnotesize, align=center] at ([xshift=-1.5cm, yshift=-0.3cm]b1-south)
    (sstlbl) {\emph{SSTable}\\\emph{(\mytilde10\,MB)}};
  \draw[-{Stealth}, thick, densely dashed, black!70, shorten >=0.1cm] (sstlbl.north) -- (b1-c3.west);

  \coordinate (t2) at ([xshift=\bucketSep]b1-north);
  \PlacePicByAnchor{bucket={28-55/23-52/25-50}}{b2}{t2}{north}
  \node[below=0.2cm of b2-south, font=\sffamily\small] {Partition 1};

  \coordinate (t3) at ([xshift=\bucketSep]b2-north);
  \PlacePicByAnchor{bucket={55-72/52-79/50-75}}{b3}{t3}{north}
  \node[below=0.2cm of b3-south, font=\sffamily\small] {Partition 2};

  \coordinate (t4) at ([xshift=\bucketSep]b3-north);
  \PlacePicByAnchor{bucket={72-103/79-103/75-100}}{b4}{t4}{north}
  \node[below=0.2cm of b4-south, font=\sffamily\small] (p3lbl) {Partition 3};

  % === Top App Box ===
  \def\appBoxMargin{1cm}
  \path let \p1 = ($(b4-center) - (b1-center)$) in
    node[draw, very thick, fill=carp0app, minimum width={\x1 + \appBoxMargin},
         minimum height=0.8cm, font=\sffamily\small\bfseries, anchor=south]
      at ($(b1-north)!0.5!(b4-north) + (0, 0.5cm)$)
      (appbox) {CARP-Augmented Parallel App};

  % Arrows from app box to buckets
  \foreach \b in {b1,b2,b3,b4} {
    \draw[->, thick, shorten <=0.1cm, shorten >=0.1cm] (appbox.south -| \b-north) -- (\b-north);
  }

  % Highlight targeted SSTables
  \draw[qannot] (b1-c1.south west) rectangle (b1-c1.north east);
  \draw[qannot] (b1-c3.south west) rectangle (b1-c3.north east);
  \draw[qannot] (b2-c2.south west) rectangle (b2-c2.north east);

  % === Query Section ===

  \begin{scope}[local bounding box=querybox]
    % Query range box
    \node[anchor=north west, font=\sffamily\bfseries, inner sep=0pt] at ([xshift=0.4cm, yshift=-1.4cm]b1-south -| sstlbl.west) (qlbl) {Query};
    \node[draw, thick, fill=white, minimum width=1.8cm, minimum height=1cm,
          font=\sffamily\small, align=center, below=0.1cm of qlbl.south west, anchor=north west]
      (qrange) {Query Range\\23-25};

    % Label baseline (hardcoded y)
    \coordinate (lbly) at ([yshift=0.0cm]qrange.south);
    \def\qboxGap{1.0cm}

    % Processing pipeline (nested scope for fit)
    \begin{scope}[local bounding box=pipelinebox]
      % Stacked SSTables icon - west at gap from qrange.east
      \coordinate (sst-target) at ([xshift=1.5cm]qrange.east);
      \PlacePicByAnchor{sststack}{sst}{sst-target}{west}
      \node[anchor=north, font=\sffamily\footnotesize, align=center] at (sst-center |- lbly)
        {Retrieve overlapping\\SSTables};

      % Merge sort box - west at gap from sst-east
      \node[draw, thick, fill=nord14!20, minimum width=1.5cm, minimum height=0.8cm,
            anchor=west] at ([xshift=\qboxGap]sst-east)
        (merge) {};
      \node[draw, thick, fill=nord14!40, minimum width=0.5cm, minimum height=0.8cm,
            font=\sffamily\small] at (merge.center) (mergehl) {0-52};
      \node[anchor=north, font=\sffamily\footnotesize] at (merge.center |- lbly) {Merge Sort};

      % Filter box - west at gap from merge.east
      \node[draw, thick, fill=nord14!40, minimum width=1.5cm, minimum height=0.8cm,
            font=\sffamily\small, align=center, anchor=west] at ([xshift=\qboxGap]merge.east)
        (filter) {23-25};
      \node[anchor=north, font=\sffamily\footnotesize, align=center] at (filter.center |- lbly)
        {Filter relevant\\data};

      % Arrows connecting query flow
      \draw[qarrow] (sst-east) -- (merge.west);
      \draw[qarrow, shorten <=0] (mergehl.east) -- (filter.west);
    \end{scope}

    % Arrow from qrange to pipeline
    \draw[qarrow,shorten <=0.1cm] (qrange.east) -- (sst-west);
  \end{scope}

  % Pipeline background
  \begin{scope}[on background layer]
    \node[draw, thick, rounded corners=5pt, fill=nord9!10,
          fit=(pipelinebox), inner sep=8pt] {};
  \end{scope}

  % Merging arrows from highlighted SSTables to sst-north
  \coordinate (junction) at ($(b1-north)!0.5!(b2-north) + (0,-1.4cm)$);
  \coordinate (ctrl) at ([yshift=-1cm]junction);
  \draw[qannot, rounded corners=3pt] (b1-c1.east) -| (junction);
  \draw[qannot, rounded corners=3pt] (b1-c3.east) -| (junction);
  \draw[qannot, rounded corners=3pt] (b2-c2.west) -| (junction);
  \draw[qannot, -{Stealth}] (junction) .. controls (ctrl) .. (sst-north);
\end{tikzpicture}
