\documentclass[border=5pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{positioning, shapes.geometric, calc, arrows.meta, fit, backgrounds, intersections}

\pgfdeclarelayer{deepbackground}
\pgfsetlayers{deepbackground,background,main}

\input{assets}

% Merge layers
\pgfsetlayers{picbackground,deepbackground,background,main}

\begin{document}

\begin{tikzpicture}[
    % Node styles
    box/.style={
      rectangle, draw=black, thick,
      minimum width=2cm, minimum height=0.8cm,
      font=\sffamily\small, align=center
    },
    decision/.style={
      diamond, draw=black, thick,
      font=\sffamily\small, align=center,
      aspect=2,
      inner sep=1pt,
      fill=carp0inner
    },
    action/.style={ box, fill=carp0inner },
    trigger/.style={ decision, fill=carp0trig },
    trigger-alt/.style={ decision, fill=carp0trig },
    appbox/.style={ box, fill=carp0app },
    process/.style={ box, fill=carp0inner },
    external/.style={ box, fill=white },
    shuffle/.style={ box, fill=carp0shuf },
    label-only/.style={ font=\sffamily\small\itshape },
    % Line styles
    ccaseline/.style={ -{Stealth}, thick },
    oobline/.style={ -{Stealth}, thick, dotted },
    renegline/.style={ -{Stealth}, thick, dashed },
  ]

  % === Decision diamond (anchor element) ===
  \node[decision] (oob-check) {Is key\\out-of-bounds\\(OOB)?};

  % === External: App I/O (top) ===
  \node[external, above=1.0cm of oob-check] (appio) {App I/O};

  % === Arrows: App I/O to OOB check ===
  % Define the diamond's top edge as a path
  \path[name path=diamond top] (oob-check.west) -- (oob-check.north) -- (oob-check.east);
  % Define the three parallel vertical lines
  \path[name path=arrow center] (appio.south) -- ++(0,-3cm);
  \path[name path=arrow left] ([xshift=-6pt]appio.south) -- ++(0,-3cm);
  \path[name path=arrow right] ([xshift=6pt]appio.south) -- ++(0,-3cm);
  % Find intersections and draw arrows
  \draw[ccaseline, name intersections={of=arrow center and diamond top, by=int-c}]
    (appio.south) -- (int-c) node[pos=0.8, right=6pt, font=\sffamily\small\itshape] (write-intercepted) {Write Intercepted};
  \draw[oobline, name intersections={of=arrow left and diamond top, by=int-l}]
    ([xshift=-6pt]appio.south) -- (int-l);
  \draw[renegline, name intersections={of=arrow right and diamond top, by=int-r}]
    ([xshift=6pt]appio.south) -- (int-r);

  % === Left branch (Yes - OOB path) ===
  \node[action, below left=0.0cm and 1.2cm of oob-check] (oob-buffer) {Buffer In\\OOB Buffer};
  \node[trigger, below=0.5cm of oob-buffer] (oob-full) {Trigger:\\Is OOB\\Buffer Full?};

  % === Right branch (No - common path) ===
  \node[trigger-alt, below right=0.2cm and 1.2cm of oob-check] (rebal-check) {Trigger:\\Rebalancing\\required?};

  % === Renegotiation (at perpendicular intersection of oob-check and oob-full) ===
  \node[process] at (oob-check |- oob-full) (reneg) {Renegotiation};

  % === Buffer Write (below left trigger) ===
  \node[appbox, fill=white, below=0.8cm of oob-full] (buffer-write) {Buffer Write};

  % === External: Shuffle Sender (bottom) ===
  \node[shuffle, below=2.8cm of reneg] (shuffle) {Shuffle Sender};

  % === Arrows from OOB check ===
  % Yes (OOB) -> Buffer In OOB Buffer
  \draw[oobline] (oob-check.west) -| (oob-buffer.north)
    node[pos=0.1, above=2pt, font=\sffamily\small\itshape] {Yes};
  % No (common + rebalancing) -> Rebalancing check
  \draw[ccaseline] (oob-check.east) -| (rebal-check.north)
    node[pos=0.1, above=4pt, font=\sffamily\small\itshape] {No};
  \draw[renegline] ([yshift=4pt]oob-check.east) -| ([xshift=4pt]rebal-check.north);

  % === Arrows: Left branch (OOB path) ===
  % Buffer In OOB Buffer -> Trigger: Is OOB Buffer Full?
  \draw[oobline] (oob-buffer.south) -- (oob-full.north);
  % Trigger: Is OOB Buffer Full? -> Renegotiation (Yes)
  \draw[oobline] (oob-full.east) -- (reneg)
    node[pos=0.1, above=2pt, font=\sffamily\small\itshape] {Yes};
  % Trigger: Is OOB Buffer Full? -> Buffer Write (No)
  \draw[oobline] (oob-full.south) -- (buffer-write.north)
    node[midway, left=2pt, font=\sffamily\small\itshape] {No};

  % === Arrows: Right branch (Rebalancing path) ===
  % Trigger: Rebalancing required? -> Renegotiation (Yes, dashed)
  \draw[renegline] (rebal-check.south) |- (reneg)
    node[pos=0.1, left=2pt, font=\sffamily\small\itshape] {Yes};
  % Trigger: Rebalancing required? -> exits right (No, solid)
  \draw[ccaseline] (rebal-check.east) -- ++(0.1cm, 0) 
    node[pos=0.5, above=2pt, font=\sffamily\small\itshape] {No}
    -- ++(0, -2.25cm) -| ([xshift=6pt]shuffle.north);

  % === Arrows to Shuffle Sender ===
  % Buffer Write -> Shuffle Sender (dotted)
  \draw[oobline] (buffer-write.east) -- ++(0.7cm, 0) 
  |- ([yshift=-4pt]reneg.west)
  node[pos=0.2, sloped, above, font=\sffamily\small\itshape] {Drain}
  ;
  % Renegotiation -> Shuffle Sender (dotted)
  \draw[oobline] ([xshift=-6pt]reneg.south) -- ([xshift=-6pt]shuffle.north);
  \draw[renegline] (reneg.south) -- (shuffle.north);

  % === Legend ===
  \begin{scope}[local bounding box=legend, shift={($(rebal-check.east) + (-4.0cm, -3.4cm)$)}]
    \node[anchor=west, font=\sffamily\small] at (0.9, 0.5) {Common Case};
    \draw[ccaseline] (0, 0.5) -- (0.8, 0.5);

    \node[anchor=west, font=\sffamily\small] at (0.9, 0) {Out-Of-Bounds};
    \draw[oobline] (0, 0) -- (0.8, 0);

    \node[anchor=west, font=\sffamily\small] at (0.9, -0.5) {Rebalancing Required};
    \draw[renegline] (0, -0.5) -- (0.8, -0.5);
  \end{scope}
  \begin{scope}[on background layer]
    \node[draw=black!60, rounded corners=3pt, fill=nord6, fit=(legend), inner xsep=4pt,inner ysep=2pt] (legend-box) {};
  \end{scope}
  \begin{pgfonlayer}{deepbackground}
    \node[draw=black!60, rounded corners=3pt, fill=carp0carp, fit=(buffer-write)(legend-box)(write-intercepted)(oob-full), inner sep=2pt] {};
  \end{pgfonlayer}

\end{tikzpicture}

\end{document}
