% ============================================================
%  MOTIF: rank (CPU / chip)
% ============================================================
\tikzset{
  pics/rank/.style args={#1}{
    code={
      % Package body
      \draw[thick, fill=gray!8, rounded corners=1pt]
        (-0.5,-0.5) rectangle (0.5,0.5);
      % Die
      \draw[semithick, fill=gray!20]
        (-0.22,-0.22) rectangle (0.22,0.22);
      % Label
      \node[anchor=center, font=\sffamily\scriptsize, inner sep=0pt] at (0,0) {#1};
      % Pins -- top & bottom
      \foreach \x in {-0.3, -0.1, 0.1, 0.3} {
        \draw[thin] (\x, 0.5) -- (\x, 0.62);
        \draw[thin] (\x,-0.5) -- (\x,-0.62);
      }
      % Pins -- left & right
      \foreach \y in {-0.3, -0.1, 0.1, 0.3} {
        \draw[thin] (-0.5,\y) -- (-0.62,\y);
        \draw[thin] ( 0.5,\y) -- ( 0.62,\y);
      }
      % Anchors
      \coordinate (-north)  at (0, 0.62);
      \coordinate (-south)  at (0,-0.62);
      \coordinate (-east)   at ( 0.62, 0);
      \coordinate (-west)   at (-0.62, 0);
      \coordinate (-center) at (0, 0);
    }
  },
  pics/rank/.default={}
}

\newcommand{\distribcurve}{
  (-0.8,0)
  .. controls (-0.65,0.05) and (-0.55,0.45) .. (-0.4,0.55)
  .. controls (-0.3,0.6) and (-0.2,0.4) .. (-0.1,0.3)

  .. controls (0.55,0.1) and (0.65,0.05) .. (0.8,0.02)

% ============================================================
%  MOTIF: orcatree (TBON-style tree: 1 root, 2 children, 4 leaves)
% ============================================================
\tikzset{
  orcatreenode/.style={
    circle, draw, thick, inner sep=1.2pt, minimum size=0.4cm
  },
  pics/orcatree/.style={
    code={
      \begin{scope}[
        grow=up,
        level distance=0.55cm,
        sibling distance=2.0cm,
        every node/.style={orcatreenode, fill=ctlcolor},
        level 1/.style={sibling distance=0.9cm, nodes={fill=aggcolor}},
        level 2/.style={sibling distance=0.45cm, nodes={fill=mpicolor}},
        edge from parent/.style={draw, thick}
      ]
        \node[orcatreenode] (root) {}
          child { node[orcatreenode] (c1) {}
            child { node[orcatreenode] (l1) {} }
            child { node[orcatreenode] (l2) {} }
          }
          child { node[orcatreenode] (c2) {}
            child { node[orcatreenode] (l3) {} }
            child { node[orcatreenode] (l4) {} }
          };
      \end{scope}

      % Anchors from bounding box
      \path (current bounding box.north)  coordinate (-north);
      \path (current bounding box.south)  coordinate (-south);
      \path (current bounding box.east)   coordinate (-east);
      \path (current bounding box.west)   coordinate (-west);
      \path (current bounding box.center) coordinate (-center);
    }
  }
}

% ============================================================
%  MOTIF: pivottable (partition boundary table)
% ============================================================
\tikzset{
  pics/pivottable/.style={
    code={
      \def\cw{1.0}  % cell width
      \def\ch{0.35} % cell height
      % Header row (Rank labels)
      \foreach \i/\lbl in {0/Rank 0, 1/Rank 1, 2/Rank 2, 3/Rank 3} {
        \draw[thick, fill=gray!15]
          ({\i*\cw - 2*\cw}, \ch) rectangle ({(\i+1)*\cw - 2*\cw}, {2*\ch});
        \node[anchor=center, font=\sffamily\scriptsize] at ({(\i+0.5)*\cw - 2*\cw}, {1.5*\ch}) {\lbl};
      }
      % Value row (ranges)
      \foreach \i/\val in {0/{0--0.3}, 1/{0.3--0.4}, 2/{0.4--0.6}, 3/{0.6--1.0}} {
        \draw[thick, fill=white]
          ({\i*\cw - 2*\cw}, 0) rectangle ({(\i+1)*\cw - 2*\cw}, \ch);
        \node[anchor=center, font=\sffamily\scriptsize] at ({(\i+0.5)*\cw - 2*\cw}, {0.5*\ch}) {\val};
      }
      % Anchors
      \coordinate (-north)  at (0, {2*\ch});
      \coordinate (-south)  at (0, 0);
      \coordinate (-east)   at ({2*\cw}, {\ch});
      \coordinate (-west)   at ({-2*\cw}, {\ch});
      \coordinate (-center) at (0, {\ch});
    }
  }
}

% ============================================================
%  MOTIF: pivottablemini (compact partition table)
% ============================================================
\tikzset{
  pics/pivottablemini/.style={
    code={

      \def\ch{0.35} % cell height
