% Lighter heading for front matter sections (Acknowledgments, Abstract)
\newcommand{\frontchapter}[1]{%
  \clearpage
  \thispagestyle{plain}% no header, just page number
  \begin{center}
    {\Large\bfseries #1}
  \end{center}
  \mtcaddchapter{} % make sure minitoc knows
  \addcontentsline{toc}{chapter}{#1}
  \vspace{2em}
}
